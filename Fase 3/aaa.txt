-- ======================
-- Extensiones + esquema
-- ======================
CREATE EXTENSION IF NOT EXISTS pgcrypto;
CREATE EXTENSION IF NOT EXISTS btree_gist;

CREATE SCHEMA IF NOT EXISTS vecindApp;
SET search_path TO vecindApp, public;

-- ======================
-- Territorio
-- ======================
CREATE TABLE IF NOT EXISTS region (
  id_region   BIGSERIAL PRIMARY KEY,
  nombre      TEXT NOT NULL UNIQUE
);

CREATE TABLE IF NOT EXISTS comuna (
  id_comuna   BIGSERIAL PRIMARY KEY,
  id_region   BIGINT NOT NULL REFERENCES region(id_region) ON DELETE RESTRICT,
  nombre      TEXT NOT NULL,
  CONSTRAINT ux_comuna UNIQUE (id_region, nombre)
);

-- ======================
-- Junta
-- ======================
CREATE TABLE IF NOT EXISTS junta (
  id_junta      BIGSERIAL PRIMARY KEY,
  id_comuna     BIGINT REFERENCES comuna(id_comuna) ON DELETE SET NULL,
  nombre        TEXT NOT NULL,
  direccion     TEXT,
  telefono      TEXT,
  email         TEXT,
  descripcion   TEXT,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- ======================
-- Seguridad: usuarios y roles
-- ======================
CREATE TABLE IF NOT EXISTS rol (
  id_rol  BIGSERIAL PRIMARY KEY,
  codigo  TEXT NOT NULL UNIQUE CHECK (codigo IN ('vecino','directiva','admin')),
  nombre  TEXT NOT NULL,
  descripcion TEXT
);

CREATE TABLE IF NOT EXISTS usuario (
  id_usuario    BIGSERIAL PRIMARY KEY,
  id_junta      BIGINT NOT NULL REFERENCES junta(id_junta) ON DELETE CASCADE,
  email         TEXT NOT NULL,
  pass_hash     TEXT NOT NULL,
  activo        BOOLEAN NOT NULL DEFAULT TRUE,
  ultimo_acceso TIMESTAMPTZ,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  CONSTRAINT ux_usuario_email UNIQUE (id_junta, email)
);

CREATE TABLE IF NOT EXISTS usuario_rol (
  id_usuario BIGINT NOT NULL REFERENCES usuario(id_usuario) ON DELETE CASCADE,
  id_rol     BIGINT NOT NULL REFERENCES rol(id_rol) ON DELETE RESTRICT,
  PRIMARY KEY (id_usuario, id_rol)
);

-- Semilla de roles (idempotente)
INSERT INTO rol (codigo, nombre, descripcion)
SELECT * FROM (VALUES
  ('vecino','Vecino','Acceso a beneficios'),
  ('directiva','Directiva','Miembro de la directiva'),
  ('admin','Administrador','Administración avanzada')
) AS t(codigo,nombre,descripcion)
ON CONFLICT (codigo) DO NOTHING;

-- ======================
-- Vecino
-- ======================
CREATE TABLE IF NOT EXISTS vecino (
  id_vecino       BIGSERIAL PRIMARY KEY,
  id_junta        BIGINT NOT NULL REFERENCES junta(id_junta) ON DELETE CASCADE,
  id_usuario      BIGINT UNIQUE REFERENCES usuario(id_usuario) ON DELETE SET NULL, -- cuenta opcional
  nombres         TEXT NOT NULL,
  apellidos       TEXT NOT NULL,
  fecha_nacimiento DATE,
  telefono        TEXT,
  email           TEXT,
  direccion       TEXT,
  id_comuna       BIGINT REFERENCES comuna(id_comuna) ON DELETE SET NULL,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- ======================
-- Espacios y Reservas
-- ======================
CREATE TABLE IF NOT EXISTS espacio (
  id_espacio  BIGSERIAL PRIMARY KEY,
  id_junta    BIGINT NOT NULL REFERENCES junta(id_junta) ON DELETE CASCADE,
  nombre      TEXT NOT NULL,
  tipo        TEXT NOT NULL CHECK (tipo IN ('cancha','sala','plaza','otro')),
  capacidad   INTEGER,
  activo      BOOLEAN NOT NULL DEFAULT TRUE
);

CREATE TABLE IF NOT EXISTS reserva (
  id_reserva   BIGSERIAL PRIMARY KEY,
  id_junta     BIGINT NOT NULL REFERENCES junta(id_junta) ON DELETE CASCADE,
  id_espacio   BIGINT NOT NULL REFERENCES espacio(id_espacio) ON DELETE CASCADE,
  id_vecino    BIGINT NOT NULL REFERENCES vecino(id_vecino) ON DELETE CASCADE,
  creado_por   BIGINT NOT NULL REFERENCES usuario(id_usuario) ON DELETE RESTRICT,
  inicio       TIMESTAMPTZ NOT NULL,
  fin          TIMESTAMPTZ NOT NULL,
  estado       TEXT NOT NULL DEFAULT 'pendiente'
               CHECK (estado IN ('pendiente','pagada','aprobada','rechazada','cancelada','confirmada')),
  observaciones TEXT,
  created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
  CONSTRAINT ck_reserva_intervalo CHECK (fin > inicio)
);
CREATE INDEX IF NOT EXISTS ix_reserva_estado ON reserva(id_junta, estado);
CREATE INDEX IF NOT EXISTS ix_reserva_espacio_tiempo ON reserva(id_espacio, inicio, fin);

-- Evitar solapes (robusto para cualquier equipo)
ALTER TABLE reserva DROP CONSTRAINT IF EXISTS ex_reserva_solape;
ALTER TABLE reserva
  ADD CONSTRAINT ex_reserva_solape
  EXCLUDE USING gist (
    id_espacio gist_int8_ops WITH =,
    tstzrange(inicio, fin, '[)') WITH &&
  )
  WHERE (estado IN ('aprobada','confirmada','pagada'))
  DEFERRABLE INITIALLY IMMEDIATE;

-- ======================
-- Certificados (sin aprobación humana)
-- ======================
CREATE TABLE IF NOT EXISTS certificado_pedido (
  id_pedido   BIGSERIAL PRIMARY KEY,
  id_junta    BIGINT NOT NULL REFERENCES junta(id_junta) ON DELETE CASCADE,
  id_vecino   BIGINT NOT NULL REFERENCES vecino(id_vecino) ON DELETE CASCADE,
  creado_por  BIGINT NOT NULL REFERENCES usuario(id_usuario) ON DELETE RESTRICT, -- quién inició
  estado      TEXT NOT NULL DEFAULT 'iniciado'
              CHECK (estado IN ('iniciado','pagado','emitido','rechazado')),
  created_at  TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX IF NOT EXISTS ix_cert_pedido_estado ON certificado_pedido(id_junta, estado);

CREATE TABLE IF NOT EXISTS certificado (
  id_certificado BIGSERIAL PRIMARY KEY,
  id_junta       BIGINT NOT NULL REFERENCES junta(id_junta) ON DELETE CASCADE,
  id_pedido      BIGINT UNIQUE NOT NULL REFERENCES certificado_pedido(id_pedido) ON DELETE CASCADE,
  numero         TEXT NOT NULL,
  fecha_emision  TIMESTAMPTZ NOT NULL DEFAULT now(),
  -- snapshot de dirección (se congela al emitir)
  direccion      TEXT,
  comuna         TEXT,
  region         TEXT,
  pdf_url        TEXT,
  CONSTRAINT ux_cert_num UNIQUE (id_junta, numero)
);

-- ======================
-- Pagos online
-- ======================
CREATE TABLE IF NOT EXISTS transaccion (
  id_transaccion BIGSERIAL PRIMARY KEY,
  id_junta       BIGINT NOT NULL REFERENCES junta(id_junta) ON DELETE CASCADE,
  id_usuario     BIGINT NOT NULL REFERENCES usuario(id_usuario) ON DELETE RESTRICT,
  origen_tipo    TEXT NOT NULL CHECK (origen_tipo IN ('certificado_pedido','reserva')),
  origen_id      BIGINT NOT NULL,   -- id del pedido o la reserva
  proveedor      TEXT NOT NULL CHECK (proveedor IN ('webpay','mercadopago','paypal')),
  monto          NUMERIC(14,2) NOT NULL CHECK (monto >= 0),
  moneda         TEXT NOT NULL DEFAULT 'CLP',
  estado         TEXT NOT NULL CHECK (estado IN ('iniciado','autorizado','rechazado','anulado')),
  external_id    TEXT,
  creado_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  autorizado_at  TIMESTAMPTZ,
  anulado_at     TIMESTAMPTZ
);
CREATE INDEX IF NOT EXISTS ix_tx_origen ON transaccion(id_junta, origen_tipo, origen_id);

CREATE TABLE IF NOT EXISTS pago_externo (
  id_pago_externo BIGSERIAL PRIMARY KEY,
  id_transaccion  BIGINT NOT NULL REFERENCES transaccion(id_transaccion) ON DELETE CASCADE,
  codigo_respuesta TEXT,
  estado_pago      TEXT,
  payload          JSONB,
  fecha_respuesta  TIMESTAMPTZ
);
CREATE UNIQUE INDEX IF NOT EXISTS ux_pagoext_tx ON pago_externo(id_transaccion);

CREATE TABLE IF NOT EXISTS detalle_transaccion (
  id_detalle     BIGSERIAL PRIMARY KEY,
  id_transaccion BIGINT NOT NULL REFERENCES transaccion(id_transaccion) ON DELETE CASCADE,
  etiqueta       TEXT,
  valor          TEXT,
  created_at     TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- ======================
-- Trigger: emitir certificado automáticamente al autorizar pago
-- ======================
CREATE OR REPLACE FUNCTION vecindApp.emitir_certificado_si_pagado()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
  IF NEW.estado = 'autorizado' AND NEW.origen_tipo = 'certificado_pedido' THEN
    -- crea certificado si no existe aún
    IF NOT EXISTS (SELECT 1 FROM vecindApp.certificado c WHERE c.id_pedido = NEW.origen_id) THEN
      INSERT INTO vecindApp.certificado (id_junta, id_pedido, numero, direccion, comuna, region, pdf_url)
      SELECT
        cp.id_junta,
        cp.id_pedido,
        CONCAT('CERT-', cp.id_pedido, '-', EXTRACT(EPOCH FROM now())::bigint),
        v.direccion,
        co.nombre,
        r.nombre,
        NULL
      FROM vecindApp.certificado_pedido cp
      JOIN vecindApp.vecino v      ON v.id_vecino = cp.id_vecino
      LEFT JOIN vecindApp.comuna co ON co.id_comuna = v.id_comuna
      LEFT JOIN vecindApp.region r  ON r.id_region = co.id_region
      WHERE cp.id_pedido = NEW.origen_id;

      UPDATE vecindApp.certificado_pedido SET estado = 'emitido' WHERE id_pedido = NEW.origen_id;
    END IF;
  END IF;
  RETURN NEW;
END $$;

DROP TRIGGER IF EXISTS trg_emitir_certificado ON vecindApp.transaccion;
CREATE TRIGGER trg_emitir_certificado
AFTER UPDATE OF estado ON vecindApp.transaccion
FOR EACH ROW
WHEN (OLD.estado IS DISTINCT FROM NEW.estado)
EXECUTE FUNCTION vecindApp.emitir_certificado_si_pagado();
